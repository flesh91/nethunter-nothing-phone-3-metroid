# Експорти, які Google вирізає (TRIM_UNUSED_KSYMS), а нашому ядру NetHunter треба лишити для USB
# Wi-Fi-драйверів із конвеєра модулів (config/nh-kernel-only.list).
# build-gki.sh дописує їх до abi_symbollist.raw. Це лише НОВІ експорти: усі стокові символи і їхні CRC
# не змінюються (перевіряє kmi-check.py), тож вендорські й гуглівські модулі вантажаться як і раніше.
# Симулятор check_gki_access.py показує, чого бракує на стоковому ядрі («NOT EXPORTED on the phone»).

# vmlinux (drivers/usb/core/urb.c)
# carl9170 (Atheros AR9170). Її usb_check_int_endpoints у vmlinux уже експортовано — на стоку
# вона лише «protected», а в нашому ядрі MODULE_SIG_PROTECT вимкнено.
usb_poison_anchored_urbs
usb_unpoison_anchored_urbs
# carl9170 і zd1211rw (ZyDAS ZD1211/ZD1211B)
usb_wait_anchor_empty_timeout

# usbnet.ko — у гуглівській копії цих експортів немає, тож на ядрі NetHunter usbnet вантажиться
# нашою збіркою замість гуглівської (як can, cdc_ncm, nfc; див. nh-gki)
# rndis_wlan (Broadcom 4320 USB, RNDIS)
usbnet_pause_rx
usbnet_purge_paused_rxq
usbnet_resume_rx

# rndis_host.ko (наш, gki/) — у нашій збірці ядра немає жодного модуля, що їх використовує, тож
# TRIM_UNUSED_KSYMS вирізав би й ці експорти; потрібні rndis_wlan
generic_rndis_bind
rndis_command
rndis_rx_fixup
rndis_status
rndis_tx_fixup
rndis_unbind
